cmake_minimum_required(VERSION 3.20)
project(Clippy VERSION 0.1.0 LANGUAGES CXX)
add_executable(Clippy WIN32 src/main.cpp src/capture.cpp src/chrome.cpp src/settings.cpp)
target_compile_features(Clippy PRIVATE cxx_std_20)
target_compile_definitions(Clippy PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(Clippy PRIVATE user32 gdi32 shell32 ole32 oleaut32 uuid uiautomationcore advapi32 comctl32 dwmapi shlwapi uxtheme)
if(MSVC)
  set_property(TARGET Clippy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(Clippy PRIVATE /W4 /utf-8)
else()
  target_compile_options(Clippy PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(Clippy PRIVATE -static -municode)
endif()
